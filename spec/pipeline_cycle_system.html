⚠ Asymmetric — có chủ đích
Tụt lùi dễ hơn tiến bộ — phản ánh đúng thực tế lâm sàng.
Tiến bộ tối đa (+2) = +0.40  |  Tụt lùi tối đa (−2) = −0.50
→ Cần nhiều session tốt để bù 1 session xấu. Không có floor — current có thể âm nếu regression nặng.
💡 Session đầu tiên trong cycle
Therapist tự nhập đánh giá so với baseline trực tiếp.
Session 1 là special case — không có "lần trước" → therapist quan sát và so sánh với trạng thái baseline đã biết.

# ── STEP 0: SCALE MAP (Asymmetric — có chủ đích) ─────────
LOCAL_TO_DELTA = {
    -2: -0.50,  # tệ hơn rõ — tụt lùi dễ hơn tiến bộ
    -1: -0.20,  # hơi tệ hơn
     0:  0.00,  # tương đương
    +1:  0.20,  # tốt hơn
    +2:  0.40,  # tốt hơn rõ rệt
}

# ── STEP 1: NORMALIZE LOCAL → DELTA ──────────────────────
def compute_block_delta(activity):
    """
    activity = { block, target_delta, local_score }
    target_delta: khoảng cần đạt (target - baseline)
    local_score: −2 → +2 (so với session gần nhất)
    """
    normalized = LOCAL_TO_DELTA[activity["local_score"]]
    return activity["target_delta"] * normalized

# ── STEP 2: AGGREGATE DELTA ──────────────────────────────
def aggregate_deltas(daily_sessions):
    """Cộng dồn delta qua mọi session trong cycle"""
    block_delta = {}
    for session in daily_sessions:
        for act in session["activities"]:
            b = act["block"]
            d = compute_block_delta(act)
            block_delta[b] = block_delta.get(b, 0) + d
    return block_delta

# ── STEP 3: CURRENT STATE ────────────────────────────────
def compute_current_blocks(baseline_blocks, block_delta):
    """
    current = baseline + cumulative_delta
    KHÔNG có floor — có thể âm nếu regression nặng
    Regression = thông tin lâm sàng quan trọng, không ẩn đi
    """
    current = {}
    for block, base_val in baseline_blocks.items():
        current[block] = base_val + block_delta.get(block, 0)
    return current

# ── STEP 4: CLAMP TRẦN (không vượt target) ───────────────
def clamp_to_target(current, target_blocks):
    """Chặn trần tại target — không clamp sàn"""
    for block in current:
        if block in target_blocks:
            current[block] = min(current[block], target_blocks[block])
    return current

# ── STEP 5: PROGRESS % ───────────────────────────────────
def compute_progress_percent(baseline_blocks, current_blocks, target_blocks):
    """
    progress < 0  → regression so với baseline
    progress = 0  → giữ nguyên baseline
    0 < p < 1     → đang tiến bộ, chưa đạt target
    progress = 1  → đạt target (clamped)
    """
    result = {}
    for block in target_blocks:
        base = baseline_blocks.get(block, 0)
        curr = current_blocks.get(block, 0)
        tgt  = target_blocks.get(block, 0)
        if tgt == base:
            result[block] = 1.0
        else:
            result[block] = (curr - base) / (tgt - base)
    return result

# ── STEP 6: ABSOLUTE SCORE (0–6) ─────────────────────────
def absolute_score(progress_pct):
    """Hệ thống tự convert progress % → absolute score để hiển thị"""
    if   progress_pct <= -0.80: return 0  # tệ hơn nhiều
    elif progress_pct <= -0.20: return 1  # hơi tệ hơn baseline
    elif progress_pct <=  0.10: return 2  # gần như giữ nguyên
    elif progress_pct <=  0.40: return 3  # tiến bộ ít
    elif progress_pct <=  0.70: return 4  # tiến bộ
    elif progress_pct <   1.00: return 5  # tiến bộ rõ rệt
    else:                        return 6  # xử lý hoàn toàn

# ── FULL PIPELINE ────────────────────────────────────────
def compute_progress_state(cycle):
    baseline       = cycle["baseline"]
    target         = cycle["target"]
    daily_sessions = cycle["daily_sessions"]

    # 1. Tích lũy delta
    block_delta = aggregate_deltas(daily_sessions)

    # 2. Current state
    current_blocks = compute_current_blocks(
        baseline["blocks"], block_delta
    )

    # 3. Clamp trần
    current_blocks = clamp_to_target(
        current_blocks, target["blocks"]
    )

    # 4. Layer scores (engine v3)
    current_layers = compute_all_layers(current_blocks)

    # 5. Progress %
    progress = compute_progress_percent(
        baseline["blocks"], current_blocks, target["blocks"]
    )

    # 6. Absolute scores
    abs_scores = {
        block: absolute_score(pct)
        for block, pct in progress.items()
    }

    # 7. Regression flags
    regressions = {
        block: pct
        for block, pct in progress.items()
        if pct < 0
    }

    return {
        "blocks":        current_blocks,
        "layers":        current_layers,
        "block_delta":   block_delta,
        "progress":      progress,
        "absolute":      abs_scores,
        "regressions":   regressions,   # flag lâm sàng quan trọng
        "sessions_count": len(daily_sessions)
    }

-- ── CYCLES TABLE ─────────────────────────────────────────
CREATE TABLE cycles (
  id           uuid        PRIMARY KEY DEFAULT gen_random_uuid(),
  child_id     uuid        REFERENCES children(id),
  teacher_id   uuid        REFERENCES user_profiles(id),
  status       text        DEFAULT 'active',  -- active | closed
  close_reason text,                           -- completed | incident | timeout
  baseline     jsonb       NOT NULL,           -- frozen forever
  target       jsonb       NOT NULL,           -- frozen for cycle duration
  started_at   date        NOT NULL,
  ended_at     date,
  created_at   timestamptz DEFAULT NOW()
);

-- ── DAILY SESSIONS TABLE ──────────────────────────────────
CREATE TABLE daily_sessions (
  id               uuid        PRIMARY KEY DEFAULT gen_random_uuid(),
  cycle_id         uuid        REFERENCES cycles(id),
  child_id         uuid        REFERENCES children(id),
  therapist_id     uuid        REFERENCES user_profiles(id),
  date             date        NOT NULL,
  is_first_session boolean     DEFAULT FALSE,
  activities       jsonb       NOT NULL,  -- [{block, target_delta, local_score, delta}]
  notes            text,
  created_at       timestamptz DEFAULT NOW()
);

-- ── RLS: chỉ GV của cycle mới được insert session ─────────
CREATE POLICY "teacher_inserts_own_sessions"
  ON daily_sessions FOR INSERT
  WITH CHECK (
    therapist_id = auth.uid()
    AND cycle_id IN (
      SELECT id FROM cycles
      WHERE teacher_id = auth.uid()
      AND status = 'active'
    )
  );

-- ── PREVENT BASELINE UPDATE sau khi lock ─────────────────
CREATE POLICY "baseline_is_immutable"
  ON cycles FOR UPDATE
  USING (status = 'active')
  WITH CHECK (
    baseline = (
      SELECT baseline FROM cycles WHERE id = cycles.id
    )
  );
